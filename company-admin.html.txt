            <div>
                <h1 class="text-2xl font-bold text-blue-400" id="companyTitleName">لوحة إدارة الشركة</h1>
                <p class="text-slate-400 text-sm">إدارة العمال، نطاق البصمة، والطلبات</p>

        <!-- إعدادات النطاق الجغرافي والعمال -->
        <div class="grid grid-cols-1 md:grid-cols-2 gap-6 mb-8">
            <div class="bg-slate-800 p-6 rounded-2xl border border-slate-700 shadow-xl">
                <h2 class="text-lg font-bold mb-4 text-blue-300">📍 تحديد نطاق العمل (Geofencing)</h2>
                <p class="text-xs text-slate-400 mb-4">حدد إحداثيات مكان العمل المسموح للموظفين البصم من داخله فقط.</p>
                <div class="space-y-3">
                    <input type="text" id="workLat" placeholder="خط العرض (Latitude)" class="w-full bg-slate-900 border border-slate-700 p-3 rounded-lg text-sm">
                    <input type="text" id="workLng" placeholder="خط الطول (Longitude)" class="w-full bg-slate-900 border border-slate-700 p-3 rounded-lg text-sm">
                    <button onclick="saveLocation()" class="w-full bg-blue-600 hover:bg-blue-700 font-bold py-2 rounded-lg text-sm">حفظ النطاق الجغرافي</button>
                </div>
            </div>

            <div class="bg-slate-800 p-6 rounded-2xl border border-slate-700 shadow-xl">
                <h2 class="text-lg font-bold mb-4 text-blue-300">👤 إضافة موظف جديد</h2>
                <div class="space-y-3">
                    <input type="text" id="empName" placeholder="اسم الموظف الثلاثي" class="w-full bg-slate-900 border border-slate-700 p-3 rounded-lg text-sm">
                    <input type="text" id="empPhone" placeholder="رقم الهاتف (رقم الموبايل)" class="w-full bg-slate-900 border border-slate-700 p-3 rounded-lg text-sm">
                    <input type="text" id="empNationalId" placeholder="رقم البطاقة الشخصية" class="w-full bg-slate-900 border border-slate-700 p-3 rounded-lg text-sm">
                    <button onclick="addEmployee()" class="w-full bg-emerald-600 hover:bg-emerald-700 font-bold py-2 rounded-lg text-sm">إضافة وتسجيل الموظف</button>
                </div>
            </div>
        </div>

        <!-- جدول الموظفين وحالة الحضور -->
        <div class="bg-slate-800 p-6 rounded-2xl border border-slate-700 shadow-xl">
            <h2 class="text-xl font-bold mb-4">قائمة العمال والموظفين المسجلين</h2>
            <div class="overflow-x-auto">
                <table class="w-full text-right text-sm">
                    <thead>
                        <tr class="border-b border-slate-700 text-slate-400">
                            <th class="p-3">اسم الموظف</th>
                            <th class="p-3">رقم الهاتف</th>
                            <th class="p-3">رقم البطاقة</th>
                            <th class="p-3">الصلاحية / الترقية</th>
                            <th class="p-3">حظر / إلغاء حظر</th>
                        </tr>
                    </thead>
                    <tbody id="employeesTable"></tbody>
                </table>
            </div>
        </div>
    </div>

    <script type="module">
        import { database, ref, set, onValue, update } from "./firebase-config.js";
        const currentCompany = localStorage.getItem('darwin_current_company');
        
        if(!currentCompany) {
            window.location.href = 'index.html';
        }

        document.getElementById('companyTitleName').innerText = "لوحة تحكم الشركة: " + currentCompany;

        window.saveLocation = function() {
            const lat = document.getElementById('workLat').value;
            const lng = document.getElementById('workLng').value;
            set(ref(database, `companiesData/${currentCompany}/settings/location`), { lat, lng }).then(() => {
                alert('تم حفظ النطاق الجغرافي بنجاح');
            });
        }

        window.addEmployee = function() {
            const name = document.getElementById('empName').value;
            const phone = document.getElementById('empPhone').value;
            const nationalId = document.getElementById('empNationalId').value;
            
            if(!name || !phone || !nationalId) { alert('أدخل كافة البيانات المطلوبة'); return; }

            const empId = 'EMP-' + Math.floor(1000 + Math.random() * 9000);
            set(ref(database, `companiesData/${currentCompany}/employees/${empId}`), {
                name, phone, nationalId, empId, role: 'employee', status: 'active'
            }).then(() => {
                alert('تم إضافة الموظف بنجاح برقم HR ID: ' + empId);
                document.getElementById('empName').value = '';
                document.getElementById('empPhone').value = '';
                document.getElementById('empNationalId').value = '';
            });
        }

        // جلب الموظفين
        onValue(ref(database, `companiesData/${currentCompany}/employees`), (snapshot) => {
            const tbody = document.getElementById('employeesTable');
            tbody.innerHTML = '';
            if(snapshot.exists()) {
                snapshot.forEach((child) => {
                    const emp = child.val();
                    tbody.innerHTML += `
                        <tr class="border-b border-slate-700/50">
                            <td class="p-3 font-bold">${emp.name}</td>
                            <td class="p-3">${emp.phone}</td>
                            <td class="p-3">${emp.nationalId}</td>
                            <td class="p-3">
                                <select onchange="updateRole('${emp.empId}', this.value)" class="bg-slate-900 border border-slate-700 p-1 rounded text-xs">
                                    <option value="employee" ${emp.role==='employee'?'selected':''}>موظف عادي</option>
                                    <option value="admin" ${emp.role==='admin'?'selected':''}>مشرف / أدمن فرعي</option>
                                </select>
                            </td>
                            <td class="p-3">
                                <button onclick="toggleBlock('${emp.empId}', '${emp.status}')" class="px-2 py-1 rounded text-xs ${emp.status==='active'?'bg-red-600':'bg-emerald-600'}">
                                    ${emp.status==='active'?'حظر':'إلغاء الحظر'}
                                </button>
                            </td>
                        </tr>
                    `;
                });
            }
        });

        window.updateRole = function(empId, role) {
            update(ref(database, `companiesData/${currentCompany}/employees/${empId}`), { role });
        }

        window.toggleBlock = function(empId, status) {
            update(ref(database, `companiesData/${currentCompany}/employees/${empId}`), { status: status === 'active' ? 'blocked' : 'active' });
        }
    </script>
